\def\pression#1#2{%
    \count0=#2
    \count1=20
    \multiply\count0 by 100
    \multiply\count1 by 100
    \advance\count0 by 27315
    \advance\count1 by 27315
    \multiply\count0 by #1
    \divide\count0 by \count1
    \count1=\count0
    \divide\count1 by 100
    \number\count1.%
    \multiply\count1 by 100
    \advance\count0 by -\count1
    \ifnum\count0<10 0\number\count0{} \else \number\count0{} \fi
}

    \pression{#1}{30}&
    \pression{#1}{35}&
    \pression{#1}{40}\\
}

\def\psi#1{%
    \count0=#1
    \multiply\count0 by 145038
    \divide\count0 by 100000
    \count1=\count0
    \divide\count1 by 10
    \number\count1.%

    \advance\count0 by -\count1
    \number\count0{}
}

\large{Remplacement}\normalsize\\
~\\
\begin{enumerate}
\item Desserrer les vis de roue
\item Lever avec un cric hydraulique
\item Poser légèrement sur une chandelle pour avoir deux points d'appui
\item Démonter la roue et dégonfler avec un embout d'une vieille pompe à pédale
\item Se changer et mettre des gants à 1 euro
\item Décoller les deux flans avec un levier en protégeant la jante
\item Lubrifier avec du liquide vaisselle et mettre le pneu au sol, face avant visible
\item Monter sur le pneu pour mettre le flan au niveau du plus petit diamètre de la jante
\item Utiliser deux barres Ken-Tool ou similaire pour commencer, bosse vers le centre
\item Continuer avec une des deux barres, bosse vers le haut, en étant toujours sur le pneu
\item Retourner et mettre le deuxième flan au niveau du plus petit diamètre
\item Utiliser une des deux barres avec la pointe contre la jante pour commencer
\item Écarter avec l'autre barre, bosse vers le haut, et pousser avec un levier
\item Couper la valve au couteau de précision et la retirer avec un embout de pompe
\item Enlever les traces de gomme de la jante avec un médiator et de l'essence
\item Mettre la nouvelle valve avec du liquide vaisselle et un embout de pompe
\item Lubrifier le nouveau pneu avec du liquide vaisselle
\item Placer le pneu sur la face avant pour être au plus proche du plus petit diamètre
\item Vérifier le sens de rotation
\item Retourner et faire passer le premier flan, bosse vers le centre
\item Retourner et faire passer le deuxième flan en terminant avec les deux barres
\item Faire coller un ou deux flans avec un compresseur ou une pompe à pied
\item Dégonfler et décoller un flan sans être proche de la valve
\item Mettre 100 \si{g} de billes d'équilibrage avec un entonnoir
\item Gonfler à 3 bars ou 44 PSI ou comme indiqué sur le pneu
\item Monter la roue avec les vis de roue graissés
\item Lever et retirer la chandelle
\item Retirer le cric
\item Serrer les vis de roue
\item Dégonfler à la pression voulue
\end{enumerate}
