\tikzstyle{carte}=[
  rectangle,
  draw = black,
  rounded corners = 3mm,
  minimum width = 63mm,
  minimum height = 88mm,
]
\tikzstyle{cadre}=[
  rectangle,
  draw = black,
  rounded corners = 2mm,
  text width = 58mm,
  align = center,
]
\newcommand\carte[4]{%
  \begin{tikzpicture}
    \node[carte] {};
    \node[cadre] (A) at (0,4) {#1\hfill#2};
    \node[cadre] (B) [below=2mm of A] {\textbf{#3}};
    \node[cadre, draw=white] (C) [below=2mm of B] {#4};
  \end{tikzpicture}%
}
